\titleformat{\section}
  {\normalfont\sffamily\Large\bfseries\color{BookText}}
  {\color{BookAccent}\thesection}
  {0.75em}
  {}
  [\vspace{0.22em}{\color{BookRule}\titlerule[0.35pt]}]
\titlespacing*{\section}{0pt}{2.15em plus 0.4em minus 0.2em}{0.85em}

\titleformat{\subsection}
  {\normalfont\sffamily\normalsize\bfseries\color{BookText}}
  {\color{BookAccent}\thesubsection}
  {0.55em}
  {}
\titlespacing*{\subsection}{0pt}{1.15em plus 0.25em minus 0.15em}{0.26em}

}
